      const logo = document.getElementById("logoBtn");

    // --- FIREBASE SYNC ---
    function fetchProducts() {
      db.ref("products").on("value", snapshot => {
        const data = snapshot.val();
        if (data) {
          const loaded = Object.keys(data).map(k => ({ id: k, ...data[k] }));
          renderProducts(loaded);
        } else {
          renderProducts(sampleProducts);
        }
      });
    }

    function addNewProduct() {
      const name = document.getElementById("admProdName").value;
      const price = Number(document.getElementById("admProdPrice").value);
      const stock = Number(document.getElementById("admProdStock").value) || 10;
      const cat = document.getElementById("admProdCat").value;
      const img = document.getElementById("admProdImg").value || "https://via.placeholder.com/150";

      if (!name || !price) return alert("Uzuza izina n'igiciro!");

      db.ref("products").push({ name, price, stock, cat, img }, err => {
        if (!err) {
          alert("Igicuruzwa cyongeweho neza!");
          document.getElementById("admProdName").value = "";
          document.getElementById("admProdPrice").value = "";
          document.getElementById("admProdImg").value = "";
        }
      });
    }

    }

      });

    // --- CART SYSTEM ---
    function addToCart(id, name, price) {
      const exist = cart.find(i => i.id === id);
      if (exist) exist.qty += 1;
      else cart.push({ id, name, price, qty: 1 });
      updateCartUI();
    }

    function updateCartUI() {
      document.getElementById("cartCountBadge").innerText = cart.reduce((a, b) => a + b.qty, 0);
      const list = document.getElementById("cartItemsList");
      list.innerHTML = "";
      let total = 0;
      cart.forEach((item, index) => {
        total += item.price * item.qty;
        list.innerHTML += `
          <div class="flex items-center justify-between bg-slate-900 p-2 rounded mb-1 text-xs">
            <div><b>${item.name}</b> x${item.qty}</div>
            <div>${(item.price * item.qty).toLocaleString()} RWF</div>
            <button onclick="removeFromCart(${index})" class="text-red-500 font-bold">✕</button>
          </div>
        `;
      });
      document.getElementById("cartTotalPrice").innerText = total.toLocaleString() + " RWF";
    }

    function removeFromCart(i) {
      cart.splice(i, 1);
      updateCartUI();
    }

    function openCartModal() {
      document.getElementById("cartModal").classList.remove("hidden");
      document.getElementById("cartModal").classList.add("flex");
    }
    function closeCartModal() {
      document.getElementById("cartModal").classList.add("hidden");
      document.getElementById("cartModal").classList.remove("flex");
    }

    // --- CHECKOUT VIA WHATSAPP ---
    function checkoutWhatsApp() {
      const name = document.getElementById("custName").value;
      const phone = document.getElementById("custPhone").value;
      const loc = document.getElementById("custLoc").value;

      if (!name || !phone || !loc || cart.length === 0) {
        return alert("Uzuza amakuru yose kandi ushyire ibicuruzwa mu gategobo!");
      }

      let orderSummary = cart.map(c => `- ${c.name} (x${c.qty}): ${(c.price * c.qty).toLocaleString()} RWF`).join("\n");
      let total = cart.reduce((a, b) => a + (b.price * b.qty), 0);

      db.ref("orders").push({
        customer: name,
        phone: phone,
        location: loc,
        items: cart,
        total: total,
        status: "Pending",
        date: new Date().toLocaleDateString()
      });

      let msg = `*ORDER MSHYA KURI OB SHOP*%0A%0A*Umuguzi:* ${name}%0A*Telefoni:* ${phone}%0A*Aho aherereye:* ${loc}%0A%0A*Ibicuruzwa:*%0A${encodeURIComponent(orderSummary)}%0A%0A*TOTAL YOSE:* ${total.toLocaleString()} RWF`;
      window.open(`https://wa.me/${TARGET_WHATSAPP}?text=${msg}`, '_blank');
    }

    function directOrder(name, price) {

      window.open(`https://wa.me/${TARGET_WHATSAPP}?text=${msg}`, '_blank');
    }

    // --- FETCH ORDERS & STATUS MANAGEMENT ---
    function fetchOrders() {
      const selectedFilter = document.getElementById("admOrderFilter").value;

      db.ref("orders").on("value", snapshot => {
        const tbody = document.getElementById("adminOrdersTableBody");
        tbody.innerHTML = "";
        let count = 0, rev = 0;

        snapshot.forEach(snap => {
          const orderId = snap.key;
          const o = snap.val();
          const currentStatus = o.status || "Pending";

          if (selectedFilter !== "All" && currentStatus !== selectedFilter) return;

          count++;
          if (currentStatus !== "Cancelled") rev += o.total || 0;

          let statusColor = "text-amber-500 bg-amber-500/10";
          if (currentStatus === "In Transit") statusColor = "text-sky-400 bg-sky-400/10";
          if (currentStatus === "Delivered") statusColor = "text-emerald-500 bg-emerald-500/10";
          if (currentStatus === "Cancelled") statusColor = "text-red-500 bg-red-500/10";

          tbody.innerHTML += `
            <tr class="border-b border-slate-700">
              <td class="p-1">${o.date || 'N/A'}</td>
              <td class="p-1"><b>${o.customer}</b><br><span class="text-gray-400">${o.phone}</span></td>
              <td class="p-1">${o.items ? o.items.length : 1} items</td>
              <td class="p-1"><b>${(o.total || 0).toLocaleString()} RWF</b></td>
              <td class="p-1"><span class="px-1.5 py-0.5 rounded font-bold ${statusColor}">● ${currentStatus}</span></td>
              <td class="p-1">
                <select onchange="updateOrderStatus('${orderId}', this.value)" class="bg-slate-900 border border-slate-700 rounded p-0.5 text-[10px]">
                  <option value="Pending" ${currentStatus === 'Pending' ? 'selected' : ''}>Pending</option>
                  <option value="In Transit" ${currentStatus === 'In Transit' ? 'selected' : ''}>In Transit</option>
                  <option value="Delivered" ${currentStatus === 'Delivered' ? 'selected' : ''}>Delivered</option>
                  <option value="Cancelled" ${currentStatus === 'Cancelled' ? 'selected' : ''}>Cancelled</option>
                </select>
              </td>
            </tr>
          `;
        });
        document.getElementById("repTotalOrders").innerText = count;
        document.getElementById("repTotalRevenue").innerText = rev.toLocaleString() + " RWF";
      });
    }

    function updateOrderStatus(orderId, newStatus) {

    }

    function openReviewModal(prodId, name) {
      selectedRevProdId = prodId;
      document.getElementById("revModalTitle").innerText = "Ibitekerezo: " + name;
      document.getElementById("reviewModal").classList.remove("hidden");
      document.getElementById("reviewModal").classList.add("flex");

      db.ref("reviews/" + prodId).on("value", snap => {
        const list = document.getElementById("reviewList");
        list.innerHTML = "";
        if (snap.exists()) {
          snap.forEach(r => {
            const rev = r.val();
            list.innerHTML += `<div class="bg-slate-900 p-2 rounded font-sans"><b>${rev.name}</b> (${'⭐'.repeat(rev.stars)})<p class="text-gray-400 mt-1">${rev.comment}</p></div>`;
          });
        } else {
          list.innerHTML = "<div class='text-gray-400'>Nta gitekerezo kirajya ku gicuruzwa.</div>";
        }
      });
    }

    function closeReviewModal() {
      document.getElementById("reviewModal").classList.add("hidden");
      document.getElementById("reviewModal").classList.remove("flex");
    }

    function submitReview() {
      const name = document.getElementById("revName").value;
      const stars = document.getElementById("revStars").value;
      const comment = document.getElementById("revComment").value;

      if (!name || !comment) return alert("Uzuza izina n'igitekerezo!");

      db.ref("reviews/" + selectedRevProdId).push({ name, stars, comment }, () => {
        document.getElementById("revName").value = "";
        document.getElementById("revComment").value = "";
      });
    }

    // --- SEARCH & CAMERA IMAGE SEARCH ---
    function searchProducts() {
      const q = document.getElementById("searchInput").value.toLowerCase();
      db.ref("products").once("value", snap => {
        const data = snap.val() ? Object.values(snap.val()) : sampleProducts;
        renderProducts(data.filter(p => p.name.toLowerCase().includes(q)));
      });
    }

    function filterCat(cat, btn) {
      document.querySelectorAll(".cat-btn").forEach(b => {
        b.classList.remove("bg-[#f59e0b]", "text-black");
        b.classList.add("bg-white/20", "text-white");
      });
      btn.classList.remove("bg-white/20", "text-white");
      btn.classList.add("bg-[#f59e0b]", "text-black");

      db.ref("products").once("value", snap => {
        const data = snap.val() ? Object.values(snap.val()) : sampleProducts;
        if (cat === "All") renderProducts(data);
        else renderProducts(data.filter(p => p.cat === cat));
      });
    }

    function triggerCamera() {
      document.getElementById("cameraInput").click();
    }

    function processCameraSearch(e) {
      if (e.target.files && e.target.files[0]) {
        const file = e.target.files[0];
        const keyword = file.name.split('.')[0].toLowerCase().replace(/[^a-z0-0]/g, '');

        db.ref("products").once("value", snap => {
          const data = snap.val() ? Object.values(snap.val()) : sampleProducts;
          let matched = data.filter(p => p.name.toLowerCase().includes(keyword));
          
          if (matched.length > 0) {
            renderProducts(matched);
            alert(`Yakiriye ifoto! Habonetse ibicuruzwa ${matched.length} bishobora gusa nayo.`);
          } else {
            renderProducts(data);
            alert("Ifoto yakiriwe! Twerekanye ibicuruzwa byose kuko nta paji ihitamo ikintu burimburi.");
          }
        });
      }
    }
